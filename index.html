📘 Практическая работа №1. Консольный AI-помощник

Цель: научиться отправлять запросы к Groq API, использовать системные промпты и обрабатывать ответы.

📁 Структура проекта

1. Настройка окружения

.env
GROQ_API_KEY=gsk_.......
assistant.py (первая версия — просто запрос)
import os
import requests
from dotenv import load_dotenv

load_dotenv()
API_KEY = os.getenv("GROQ_API_KEY")
URL = "https://api.groq.com/openai/v1/chat/completions"

def ask_groq(prompt, system_prompt=None):
    headers = {
        "Authorization": f"Bearer {API_KEY}",
        "Content-Type": "application/json"
    }
    messages = []
    if system_prompt:
        messages.append({"role": "system", "content": system_prompt})
    messages.append({"role": "user", "content": prompt})

    payload = {
        "model": "llama-3.3-70b-versatile",
        "messages": messages,
        "temperature": 0.7,
        "max_tokens": 200
    }
    response = requests.post(URL, json=payload, headers=headers)
    response.raise_for_status()
    return response.json()["choices"][0]["message"]["content"]

if __name__ == "__main__":
    print("🤖 AI помощник (напиши 'выход' для завершения)")
    system = "Ты дружелюбный эксперт по программированию на Python. Отвечай кратко и понятно."
    while True:
        user_input = input("\n🧑‍💻 Вы: ")
        if user_input.lower() in ("выход", "exit"):
            break
        answer = ask_groq(user_input, system)
        print(f"🤖 AI: {answer}")
💡 Пояснение: Мы используем библиотеку requests для POST-запроса. load_dotenv загружает ключ из файла .env — это безопаснее, чем хранить ключ в коде.

🌤️ Практическая работа №2. AI с реальными данными о погоде

Цель: объединить вызов внешнего API погоды (Open-Meteo) с Groq — AI будет отвечать на вопросы, используя реальные данные.

Как это работает

  1. Пользователь спрашивает: «Какая погода в Минске?»
  2. Программа распознаёт ключевые слова и вызывает функцию get_weather(city).
  3. Полученную погоду передаём в Groq, чтобы AI сформулировал красивый ответ.
import os, requests
from dotenv import load_dotenv

load_dotenv()
API_KEY = os.getenv("GROQ_API_KEY")

def get_weather(city):
    # Получаем координаты
    geo_url = f"https://geocoding-api.open-meteo.com/v1/search?name={city}&count=1&language=ru&format=json"
    geo_resp = requests.get(geo_url).json()
    if not geo_resp.get("results"):
        return None
    lat = geo_resp["results"][0]["latitude"]
    lon = geo_resp["results"][0]["longitude"]
    # Получаем погоду
    weather_url = f"https://api.open-meteo.com/v1/forecast?latitude={lat}&longitude={lon}¤t_weather=true"
    weather_data = requests.get(weather_url).json()
    temp = weather_data["current_weather"]["temperature"]
    wind = weather_data["current_weather"]["windspeed"]
    return f"Сейчас {temp}°C, ветер {wind} км/ч"

def ask_groq_with_weather(user_question):
    # Проверяем, не спрашивает ли пользователь о погоде
    if "погод" in user_question.lower():
        # Простейшее извлечение города (можно улучшить через регулярное выражение)
        words = user_question.split()
        city = None
        for w in words:
            if w.lower() in ("минск", "москва", "кий", "питер", "брест", "гродно"):
                city = w
                break
        if city:
            weather = get_weather(city)
            if weather:
                prompt = f"Пользователь спросил: '{user_question}'. Реальная погода в {city}: {weather}. Ответь дружелюбно, включив эти данные."
            else:
                prompt = f"Пользователь спросил о погоде, но город '{city}' не найден. Сообщи об этом вежливо."
        else:
            prompt = user_question
    else:
        prompt = user_question

    headers = {"Authorization": f"Bearer {API_KEY}", "Content-Type": "application/json"}
    payload = {
        "model": "llama-3.3-70b-versatile",
        "messages": [{"role": "user", "content": prompt}],
        "temperature": 0.7
    }
    resp = requests.post("https://api.groq.com/openai/v1/chat/completions", json=payload, headers=headers)
    return resp.json()["choices"][0]["message"]["content"]

# Пример использования
if __name__ == "__main__":
    while True:
        q = input("\nЗадайте вопрос о погоде (или другое): ")
        if q.lower() == "выход": break
        answer = ask_groq_with_weather(q)
        print(f"🤖 AI: {answer}")
✨ Важно: Эта версия не использует function calling, а просто ищет ключевые слова. В следующей работе мы реализуем более надёжный механизм через инструменты (tools) Groq API.

⏱️ Практическая работа №3. Асинхронные запросы и Function Calling

Цель: научиться использовать встроенный механизм tools/function calling Groq API для надёжного получения данных о погоде, а также освоить асинхронные запросы с aiohttp для повышения производительности.

3.1 Function Calling (синхронный вариант)

import os, json, requests
from dotenv import load_dotenv

load_dotenv()
API_KEY = os.getenv("GROQ_API_KEY")

def get_weather_info(city: str) -> str:
    """Реальная функция, вызываемая AI"""
    geo = requests.get(f"https://geocoding-api.open-meteo.com/v1/search?name={city}&count=1&format=json").json()
    if not geo.get("results"):
        return f"Город {city} не найден"
    lat, lon = geo["results"][0]["latitude"], geo["results"][0]["longitude"]
    weather = requests.get(f"https://api.open-meteo.com/v1/forecast?latitude={lat}&longitude={lon}¤t_weather=true").json()
    t = weather["current_weather"]["temperature"]
    w = weather["current_weather"]["windspeed"]
    return f"🌡️ {t}°C, ветер {w} км/ч"

# Описание инструмента для AI
weather_tool = {
    "type": "function",
    "function": {
        "name": "get_weather_info",
        "description": "Возвращает текущую погоду в городе",
        "parameters": {
            "type": "object",
            "properties": {
                "city": {"type": "string", "description": "Название города"}
            },
            "required": ["city"]
        }
    }
}

def chat_with_tools():
    messages = [{"role": "system", "content": "Ты помощник. Если спрашивают погоду, используй функцию get_weather_info."}]
    while True:
        user = input("\nВы: ")
        if user.lower() == "выход": break
        messages.append({"role": "user", "content": user})
        payload = {
            "model": "llama-3.3-70b-versatile",
            "messages": messages,
            "tools": [weather_tool],
            "tool_choice": "auto",
            "temperature": 0.7
        }
        resp = requests.post("https://api.groq.com/openai/v1/chat/completions",
                             json=payload, headers={"Authorization": f"Bearer {API_KEY}", "Content-Type": "application/json"})
        data = resp.json()
        msg = data["choices"][0]["message"]
        if msg.get("tool_calls"):
            func_call = msg["tool_calls"][0]["function"]
            args = json.loads(func_call["arguments"])
            weather_result = get_weather_info(args["city"])
            # Отправляем результат функции обратно модели
            messages.append(msg)  # сохраняем вызов функции
            messages.append({"role": "tool", "content": weather_result, "tool_call_id": msg["tool_calls"][0]["id"]})
            # Повторный запрос к модели для финального ответа
            second_resp = requests.post("https://api.groq.com/openai/v1/chat/completions",
                                        json={"model": "llama-3.3-70b-versatile", "messages": messages},
                                        headers={"Authorization": f"Bearer {API_KEY}"}).json()
            answer = second_resp["choices"][0]["message"]["content"]
        else:
            answer = msg["content"]
        print(f"🤖 AI: {answer}")
        messages.append({"role": "assistant", "content": answer})

if __name__ == "__main__":
    chat_with_tools()
🔧 Механизм: AI может вернуть tool_calls вместо обычного ответа. Ваш код перехватывает этот вызов, выполняет реальную функцию, и отправляет результат обратно модели. Так AI получает свежие данные.

3.2 Асинхронная версия (aiohttp)

import asyncio, aiohttp, os, json
from dotenv import load_dotenv

load_dotenv()
API_KEY = os.getenv("GROQ_API_KEY")

async def get_weather_async(session, city):
    async with session.get(f"https://geocoding-api.open-meteo.com/v1/search?name={city}&count=1&format=json") as resp:
        geo = await resp.json()
    if not geo.get("results"): return None
    lat, lon = geo["results"][0]["latitude"], geo["results"][0]["longitude"]
    async with session.get(f"https://api.open-meteo.com/v1/forecast?latitude={lat}&longitude={lon}¤t_weather=true") as resp:
        wdata = await resp.json()
    return f"{wdata['current_weather']['temperature']}°C, ветер {wdata['current_weather']['windspeed']} км/ч"

async def main():
    async with aiohttp.ClientSession() as session:
        while True:
            q = input("\nВаш запрос: ")
            if q == "выход": break
            # Здесь можно было бы вызвать Groq и обработать tool_calls асинхронно
            # Для простоты пока просто погода
            if "погода" in q.lower():
                city = q.split()[-1]
                weather = await get_weather_async(session, city)
                print(f"🌦️ {weather}")
            else:
                print("(Синхронный вызов Groq – вы можете расширить пример)")

if __name__ == "__main__":
    asyncio.run(main())

🤖 Практическая работа №4. LlamaBot — телеграм-бот на Python с Groq

Цель: создать полноценного Telegram-бота с использованием библиотеки python-telegram-bot и Groq API. Бот будет помнить историю разговора (по каждому пользователю отдельно) и поддерживать системный промпт.

Подготовка

  • Установите: pip install python-telegram-bot python-dotenv requests
  • Создайте бота в Telegram через @BotFather и получите токен.
  • Добавьте в .env: TELEGRAM_TOKEN=ваш_токен и GROQ_API_KEY=...
import os, json, requests
from dotenv import load_dotenv
from telegram import Update
from telegram.ext import Application, CommandHandler, MessageHandler, filters, ContextTypes

load_dotenv()
GROQ_KEY = os.getenv("GROQ_API_KEY")
TELEGRAM_TOKEN = os.getenv("TELEGRAM_TOKEN")
GROQ_URL = "https://api.groq.com/openai/v1/chat/completions"

# Хранилище истории диалогов (user_id → список сообщений)
user_sessions = {}

SYSTEM_PROMPT = "Ты весёлый и полезный AI-помощник. Отвечай кратко, добавляй эмодзи."

def get_groq_response(messages):
    headers = {"Authorization": f"Bearer {GROQ_KEY}", "Content-Type": "application/json"}
    payload = {
        "model": "llama-3.3-70b-versatile",
        "messages": messages,
        "temperature": 0.8,
        "max_tokens": 300
    }
    resp = requests.post(GROQ_URL, json=payload, headers=headers)
    resp.raise_for_status()
    return resp.json()["choices"][0]["message"]["content"]

async def start(update: Update, context: ContextTypes.DEFAULT_TYPE):
    await update.message.reply_text("🤖 Привет! Я LlamaBot на Groq API. Задавай любые вопросы! Используй /clear для сброса истории.")

async def clear(update: Update, context: ContextTypes.DEFAULT_TYPE):
    user_id = update.effective_user.id
    user_sessions[user_id] = []
    await update.message.reply_text("🧹 История диалога очищена.")

async def handle_message(update: Update, context: ContextTypes.DEFAULT_TYPE):
    user_id = update.effective_user.id
    user_text = update.message.text
    # Инициализируем историю для пользователя
    if user_id not in user_sessions:
        user_sessions[user_id] = [{"role": "system", "content": SYSTEM_PROMPT}]
    # Добавляем сообщение пользователя
    user_sessions[user_id].append({"role": "user", "content": user_text})
    # Ограничиваем длину истории (последние 10 сообщений + system)
    if len(user_sessions[user_id]) > 12:
        user_sessions[user_id] = [user_sessions[user_id][0]] + user_sessions[user_id][-10:]
    try:
        answer = get_groq_response(user_sessions[user_id])
        user_sessions[user_id].append({"role": "assistant", "content": answer})
        await update.message.reply_text(answer)
    except Exception as e:
        await update.message.reply_text(f"❌ Ошибка: {str(e)}")

def main():
    app = Application.builder().token(TELEGRAM_TOKEN).build()
    app.add_handler(CommandHandler("start", start))
    app.add_handler(CommandHandler("clear", clear))
    app.add_handler(MessageHandler(filters.TEXT & ~filters.COMMAND, handle_message))
    print("Бот запущен...")
    app.run_polling()

if __name__ == "__main__":
    main()
