📘 Практическая работа №1. Консольный AI-помощник

Цель: научиться отправлять запросы к Groq API, использовать системные промпты и обрабатывать ответы.

Базовый код (assistant.py)

import os
import requests
from dotenv import load_dotenv

load_dotenv()
API_KEY = os.getenv("GROQ_API_KEY")
URL = "https://api.groq.com/openai/v1/chat/completions"

def ask_groq(prompt, system_prompt=None):
    headers = {"Authorization": f"Bearer {API_KEY}", "Content-Type": "application/json"}
    messages = []
    if system_prompt:
        messages.append({"role": "system", "content": system_prompt})
    messages.append({"role": "user", "content": prompt})
    payload = {
        "model": "llama-3.3-70b-versatile",
        "messages": messages,
        "temperature": 0.7,
        "max_tokens": 200
    }
    response = requests.post(URL, json=payload, headers=headers)
    response.raise_for_status()
    return response.json()["choices"][0]["message"]["content"]

if __name__ == "__main__":
    print("🤖 AI помощник (напиши 'выход' для завершения)")
    system = "Ты дружелюбный эксперт по программированию на Python. Отвечай кратко и понятно."
    while True:
        user_input = input("\n🧑‍💻 Вы: ")
        if user_input.lower() in ("выход", "exit"):
            break
        answer = ask_groq(user_input, system)
        print(f"🤖 AI: {answer}")

import os, requests

load_dotenv()
API_KEY = os.getenv("GROQ_API_KEY")
URL = "https://api.groq.com/openai/v1/chat/completions"

SYSTEM = "Ты дружелюбный эксперт по Python. Отвечай кратко."
history = []
temperature = 0.7

def ask_groq(messages):
    headers = {"Authorization": f"Bearer {API_KEY}", "Content-Type": "application/json"}
    payload = {"model": "llama-3.3-70b-versatile", "messages": messages, "temperature": temperature}
    resp = requests.post(URL, json=payload, headers=headers)
    resp.raise_for_status()
    return resp.json()["choices"][0]["message"]["content"]

print("🤖 AI помощник. Команды: /history, /clear, /temp [число], /exit")
while True:
    user = input("\nВы: ")
    if user.lower() in ("/exit", "выход"):
        break
    if user == "/history":
        for m in history[-10:]:
            print(f"{m['role']}: {m['content'][:100]}")
        continue
    if user == "/clear":
        history = []
        print("История очищена.")
        continue
    if user.startswith("/temp"):
        try:
            temperature = float(user.split()[1])
            print(f"Температура установлена в {temperature}")
        except:
            print("Используйте: /temp 0.7")
        continue
    history.append({"role": "user", "content": user})
    messages = [{"role": "system", "content": SYSTEM}] + history[-10:]
    answer = ask_groq(messages)
    print(f"🤖 AI: {answer}")
    history.append({"role": "assistant", "content": answer})

Расширенный вариант: погода + советы от Groq (с .env)

import os
import requests
from dotenv import load_dotenv

load_dotenv()
GROQ_API_KEY = os.getenv("GROQ_API_KEY")

def get_weather(city):
    geo_url = f"https://geocoding-api.open-meteo.com/v1/search?name={city}&count=1&format=json"
    geo_resp = requests.get(geo_url).json()
    if not geo_resp.get("results"):
        return None, None
    lat = geo_resp["results"][0]["latitude"]
    lon = geo_resp["results"][0]["longitude"]
    weather_url = f"https://api.open-meteo.com/v1/forecast?latitude={lat}&longitude={lon}¤t_weather=true"
    w = requests.get(weather_url).json()
    temp = w["current_weather"]["temperature"]
    wind = w["current_weather"]["windspeed"]
    return temp, wind

def ask_groq(prompt):
    headers = {"Authorization": f"Bearer {GROQ_API_KEY}", "Content-Type": "application/json"}
    payload = {
        "model": "llama-3.3-70b-versatile",
        "messages": [{"role": "user", "content": prompt}],
        "temperature": 0.7,
        "max_tokens": 200
    }
    resp = requests.post("https://api.groq.com/openai/v1/chat/completions", json=payload, headers=headers)
    return resp.json()["choices"][0]["message"]["content"]

while True:
    city = input("\n🌍 Введите город на английском (или 'выход'): ").strip()
    if city.lower() in ("выход", "exit"):
        break
    temp, wind = get_weather(city)
    if temp is None:
        print("❌ Город не найден, попробуйте ещё раз.")
        continue
    print(f"\n🌡️ Погода в {city}: {temp}°C, ветер {wind} км/ч")
    prompt = f"""Погода: температура {temp}°C, ветер {wind} км/ч.
Дай краткий совет (2-3 предложения): как одеваться, нужен ли зонт/шапка, стоит ли выходить на улицу.
Ответь дружелюбно, без лишних слов."""
    advice = ask_groq(prompt)
    print(f"💡 Совет от ИИ: {advice}")

📌 Улучшенное распознавание городов (кириллица + латиница) 
city_map = {
    "минск": "Minsk", "москва": "Moscow", "киев": "Kyiv",
    "питер": "Saint Petersburg", "брест": "Brest", "гродно": "Grodno"
}
def extract_city(text):
    text_lower = text.lower()
    for ru, en in city_map.items():
        if ru in text_lower:
            return en
    return text.split()[-1].strip(",.?!")

📘 Практическая работа №3. Telegram-бот (aiogram) + Groq + курс валют НБРБ

Цель: создать асинхронного Telegram-бота на aiogram, интегрировать Groq API для умных ответов, добавить историю диалога и запрос курса валют через API Национального банка Республики Беларусь с поддержкой даты и разных валют (USD, EUR, RUB).

⚡ Почему aiogram? Асинхронный фреймворк, высокая производительность, поддержка FSM и роутеров. Стандарт для современных ботов на Python.

Базовое решение: эхо-бот (проверка токена)

import os
from dotenv import load_dotenv
from aiogram import Bot, Dispatcher, types
from aiogram.filters import Command
import asyncio

load_dotenv()
TELEGRAM_TOKEN = os.getenv("TELEGRAM_TOKEN")

bot = Bot(token=TELEGRAM_TOKEN)
dp = Dispatcher()

@dp.message(Command("start"))
async def cmd_start(message: types.Message):
    await message.answer("Привет! Я эхо-бот. Просто напиши что-нибудь.")

@dp.message()
async def echo(message: types.Message):
    await message.answer(f"Эхо: {message.text}")

async def main():
    print("Эхо-бот запущен")
    await dp.start_polling(bot)

if __name__ == "__main__":
    asyncio.run(main())

Расширенное решение: AI-бот с Groq (без истории)

import os, aiohttp
from dotenv import load_dotenv
from aiogram import Bot, Dispatcher, types
from aiogram.filters import Command
import asyncio

load_dotenv()
GROQ_API_KEY = os.getenv("GROQ_API_KEY")
TELEGRAM_TOKEN = os.getenv("TELEGRAM_TOKEN")
GROQ_URL = "https://api.groq.com/openai/v1/chat/completions"

bot = Bot(token=TELEGRAM_TOKEN)
dp = Dispatcher()

async def ask_groq(prompt):
    headers = {"Authorization": f"Bearer {GROQ_API_KEY}", "Content-Type": "application/json"}
    payload = {
        "model": "llama-3.3-70b-versatile",
        "messages": [{"role": "user", "content": prompt}],
        "temperature": 0.7,
        "max_tokens": 200
    }
    async with aiohttp.ClientSession() as session:
        async with session.post(GROQ_URL, json=payload, headers=headers) as resp:
            data = await resp.json()
            return data["choices"][0]["message"]["content"]

@dp.message(Command("start"))
async def cmd_start(message: types.Message):
    await message.answer("Я AI-бот на Groq. Задавай любые вопросы!")

@dp.message()
async def handle_message(message: types.Message):
    answer = await ask_groq(message.text)
    await message.answer(answer)

async def main():
    print("AI-бот запущен")
    await dp.start_polling(bot)

if __name__ == "__main__":
    asyncio.run(main())

📌 Полный интеграционный пример (всё вместе) 
import os, aiohttp, re
from datetime import datetime
from dotenv import load_dotenv
from aiogram import Bot, Dispatcher, types
from aiogram.filters import Command
import asyncio

load_dotenv()
GROQ_API_KEY = os.getenv("GROQ_API_KEY")
TELEGRAM_TOKEN = os.getenv("TELEGRAM_TOKEN")
GROQ_URL = "https://api.groq.com/openai/v1/chat/completions"

bot = Bot(token=TELEGRAM_TOKEN)
dp = Dispatcher()
user_sessions = {}
SYSTEM_PROMPT = "Ты дружелюбный помощник. Отвечай кратко."

async def ask_groq_with_history(user_id, text):
    if user_id not in user_sessions:
        user_sessions[user_id] = [{"role": "system", "content": SYSTEM_PROMPT}]
    user_sessions[user_id].append({"role": "user", "content": text})
    if len(user_sessions[user_id]) > 12:
        user_sessions[user_id] = [user_sessions[user_id][0]] + user_sessions[user_id][-10:]
    headers = {"Authorization": f"Bearer {GROQ_API_KEY}", "Content-Type": "application/json"}
    payload = {"model": "llama-3.3-70b-versatile", "messages": user_sessions[user_id], "temperature": 0.8}
    async with aiohttp.ClientSession() as session:
        async with session.post(GROQ_URL, json=payload, headers=headers) as resp:
            data = await resp.json()
            answer = data["choices"][0]["message"]["content"]
    user_sessions[user_id].append({"role": "assistant", "content": answer})
    return answer

async def get_rate(currency_code, date_str=None):
    if date_str is None:
        date_str = datetime.now().strftime("%Y-%m-%d")
    url = f"https://api.nbrb.by/exrates/rates/{currency_code}?parammode=2&ondate={date_str}"
    async with aiohttp.ClientSession() as session:
        async with session.get(url) as resp:
            if resp.status != 200:
                return None
            data = await resp.json()
            return {
                "rate": data["Cur_OfficialRate"],
                "scale": data["Cur_Scale"],
                "name": data["Cur_Name"],
                "date": data["Date"]
            }

@dp.message(Command("start"))
async def cmd_start(message: types.Message):
    await message.answer("Бот с Groq и курсом валют НБРБ.\n/clear – очистить историю.\nПримеры: 'курс доллара', 'курс евро на 2024-05-13'")

@dp.message(Command("clear"))
async def cmd_clear(message: types.Message):
    user_sessions[message.from_user.id] = []
    await message.answer("История очищена.")

@dp.message()
async def handle_message(message: types.Message):
    text = message.text.lower()
    user_id = message.from_user.id
    if "курс" in text:
        currency = None
        if "доллар" in text or "usd" in text:
            currency = "USD"
        elif "евро" in text or "eur" in text:
            currency = "EUR"
        elif "рубль" in text or "rub" in text:
            currency = "RUB"
        else:
            await message.answer("Напишите 'курс доллара', 'курс евро' или 'курс рубля'.\nМожно указать дату: 'курс доллара на 2024-05-13'")
            return
        date_match = re.search(r'(\d{4}-\d{2}-\d{2})', text)
        date_str = date_match.group(1) if date_match else None
        rate_data = await get_rate(currency, date_str)
        if rate_data is None:
            await message.answer(f"Не удалось получить курс {currency.upper()} на указанную дату. Проверьте правильность даты.")
            return
        scale = rate_data["scale"]
        rate = rate_data["rate"]
        name = rate_data["name"]
        date_display = rate_data["date"][:10]
        if scale == 1:
            await message.answer(f"Курс {name} на {date_display}: {rate} руб.")
        else:
            await message.answer(f"Курс {scale} {name} на {date_display}: {rate} руб. (за {scale} единиц)")
    else:
        answer = await ask_groq_with_history(user_id, message.text)
        await message.answer(answer)

async def main():
    print("Бот запущен")
    await dp.start_polling(bot)

if __name__ == "__main__":
    asyncio.run(main())
