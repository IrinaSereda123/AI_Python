📘 Практическая работа №1. Консольный AI-помощник

Цель: научиться отправлять запросы к Groq API, использовать системные промпты и обрабатывать ответы.

Базовый код (assistant.py)

import os
import requests
from dotenv import load_dotenv

load_dotenv()
API_KEY = os.getenv("GROQ_API_KEY")
URL = "https://api.groq.com/openai/v1/chat/completions"

def ask_groq(prompt, system_prompt=None):
    headers = {"Authorization": f"Bearer {API_KEY}", "Content-Type": "application/json"}
    messages = []
    if system_prompt:
        messages.append({"role": "system", "content": system_prompt})
    messages.append({"role": "user", "content": prompt})
    payload = {
        "model": "llama-3.3-70b-versatile",
        "messages": messages,
        "temperature": 0.7,
        "max_tokens": 200
    }
    response = requests.post(URL, json=payload, headers=headers)
    response.raise_for_status()
    return response.json()["choices"][0]["message"]["content"]

if __name__ == "__main__":
    print("🤖 AI помощник (напиши 'выход' для завершения)")
    system = "Ты дружелюбный эксперт по программированию на Python. Отвечай кратко и понятно."
    while True:
        user_input = input("\n🧑‍💻 Вы: ")
        if user_input.lower() in ("выход", "exit"):
            break
        answer = ask_groq(user_input, system)
        print(f"🤖 AI: {answer}")

🌤️ Практическая работа №2. AI с реальными данными о погоде

Цель: объединить вызов внешнего API погоды с Groq — AI будет отвечать на вопросы, используя реальные данные.

import os, requests
from dotenv import load_dotenv
load_dotenv()
API_KEY = os.getenv("GROQ_API_KEY")

def get_weather(city):
    geo = requests.get(f"https://geocoding-api.open-meteo.com/v1/search?name={city}&count=1&format=json").json()
    if not geo.get("results"): return None
    lat, lon = geo["results"][0]["latitude"], geo["results"][0]["longitude"]
    w = requests.get(f"https://api.open-meteo.com/v1/forecast?latitude={lat}&longitude={lon}¤t_weather=true").json()
    return f"{w['current_weather']['temperature']}°C, ветер {w['current_weather']['windspeed']} км/ч"

def ask_groq(prompt):
    headers = {"Authorization": f"Bearer {API_KEY}"}
    resp = requests.post("https://api.groq.com/openai/v1/chat/completions",
                         json={"model": "llama-3.3-70b-versatile", "messages": [{"role": "user", "content": prompt}]},
                         headers=headers)
    return resp.json()["choices"][0]["message"]["content"]

while True:
    q = input("\nВы: ")
    if q.lower() in ("exit","выход"): break
    if "погод" in q.lower():
        city = q.split()[-1]
        weather = get_weather(city)
        if weather:
            prompt = f"Пользователь спросил: '{q}'. Реальная погода в {city}: {weather}. Ответь кратко."
        else:
            prompt = f"Пользователь спросил о погоде в {city}, но город не найден. Сообщи об этом."
    else:
        prompt = q
    print(f"🤖 AI: {ask_groq(prompt)}")

⏱️ Практическая работа №3. Function Calling и асинхронные запросы

Цель: использовать механизм tools (function calling) Groq API и асинхронный HTTP.

Синхронный вариант с tools

import os, json, requests
from dotenv import load_dotenv
load_dotenv()
API_KEY = os.getenv("GROQ_API_KEY")
URL = "https://api.groq.com/openai/v1/chat/completions"

def get_weather_info(city):
    geo = requests.get(f"https://geocoding-api.open-meteo.com/v1/search?name={city}&count=1&format=json").json()
    if not geo.get("results"): return f"Город {city} не найден"
    lat, lon = geo["results"][0]["latitude"], geo["results"][0]["longitude"]
    w = requests.get(f"https://api.open-meteo.com/v1/forecast?latitude={lat}&longitude={lon}¤t_weather=true").json()
    return f"{w['current_weather']['temperature']}°C, ветер {w['current_weather']['windspeed']} км/ч"

weather_tool = {
    "type": "function",
    "function": {
        "name": "get_weather_info",
        "description": "Возвращает погоду в городе",
        "parameters": {"type": "object", "properties": {"city": {"type": "string"}}, "required": ["city"]}
    }
}

messages = [{"role": "system", "content": "Ты помощник. Если спрашивают погоду, используй функцию get_weather_info."}]
while True:
    user = input("\nВы: ")
    if user.lower() in ("выход","exit"): break
    messages.append({"role": "user", "content": user})
    payload = {"model": "llama-3.3-70b-versatile", "messages": messages, "tools": [weather_tool], "tool_choice": "auto"}
    headers = {"Authorization": f"Bearer {API_KEY}"}
    resp = requests.post(URL, json=payload, headers=headers).json()
    msg = resp["choices"][0]["message"]
    if msg.get("tool_calls"):
        args = json.loads(msg["tool_calls"][0]["function"]["arguments"])
        weather = get_weather_info(args["city"])
        messages.append(msg)
        messages.append({"role": "tool", "content": weather, "tool_call_id": msg["tool_calls"][0]["id"]})
        final = requests.post(URL, json={"model": "llama-3.3-70b-versatile", "messages": messages}, headers=headers).json()
        answer = final["choices"][0]["message"]["content"]
    else:
        answer = msg["content"]
    print(f"🤖 AI: {answer}")
    messages.append({"role": "assistant", "content": answer})

📌 Асинхронный вызов погоды (aiohttp) – полный пример 
import asyncio, aiohttp, os
from dotenv import load_dotenv
load_dotenv()
API_KEY = os.getenv("GROQ_API_KEY")

async def get_weather_async(session, city):
    async with session.get(f"https://geocoding-api.open-meteo.com/v1/search?name={city}&count=1&format=json") as resp:
        geo = await resp.json()
    if not geo.get("results"): return None
    lat, lon = geo["results"][0]["latitude"], geo["results"][0]["longitude"]
    async with session.get(f"https://api.open-meteo.com/v1/forecast?latitude={lat}&longitude={lon}¤t_weather=true") as resp:
        w = await resp.json()
    return f"{w['current_weather']['temperature']}°C, ветер {w['current_weather']['windspeed']} км/ч"

async def main():
    async with aiohttp.ClientSession() as session:
        while True:
            city = input("Город (англ): ")
            if city == "exit": break
            weather = await get_weather_async(session, city)
            print(weather if weather else "Город не найден")
asyncio.run(main())
📌 Параллельные запросы к нескольким городам 
cities = ["Minsk", "Paris", "London", "Tokyo"]
async def fetch_all():
    async with aiohttp.ClientSession() as session:
        tasks = [get_weather_async(session, city) for city in cities]
        results = await asyncio.gather(*tasks)
        for city, res in zip(cities, results):
            print(f"{city}: {res}")
asyncio.run(fetch_all())

🤖 Практическая работа №4. LlamaBot — телеграм-бот на Python с Groq

Цель: создать полноценного Telegram-бота с использованием python-telegram-bot и Groq API.

import os, requests
from dotenv import load_dotenv
from telegram import Update
from telegram.ext import Application, CommandHandler, MessageHandler, filters, ContextTypes

load_dotenv()
GROQ_KEY = os.getenv("GROQ_API_KEY")
TELEGRAM_TOKEN = os.getenv("TELEGRAM_TOKEN")
GROQ_URL = "https://api.groq.com/openai/v1/chat/completions"

user_sessions = {}
SYSTEM = "Ты весёлый помощник. Отвечай кратко, добавляй эмодзи."

def get_groq_response(messages):
    headers = {"Authorization": f"Bearer {GROQ_KEY}"}
    payload = {"model": "llama-3.3-70b-versatile", "messages": messages, "temperature": 0.8}
    resp = requests.post(GROQ_URL, json=payload, headers=headers)
    return resp.json()["choices"][0]["message"]["content"]

async def start(update, context):
    await update.message.reply_text("🤖 Привет! Я LlamaBot. /clear – очистить историю.")

async def clear(update, context):
    user_sessions[update.effective_user.id] = []
    await update.message.reply_text("🧹 История очищена.")

async def handle_message(update, context):
    user_id = update.effective_user.id
    text = update.message.text
    if user_id not in user_sessions:
        user_sessions[user_id] = [{"role": "system", "content": SYSTEM}]
    user_sessions[user_id].append({"role": "user", "content": text})
    if len(user_sessions[user_id]) > 12:
        user_sessions[user_id] = [user_sessions[user_id][0]] + user_sessions[user_id][-10:]
    answer = get_groq_response(user_sessions[user_id])
    user_sessions[user_id].append({"role": "assistant", "content": answer})
    await update.message.reply_text(answer)

def main():
    app = Application.builder().token(TELEGRAM_TOKEN).build()
    app.add_handler(CommandHandler("start", start))
    app.add_handler(CommandHandler("clear", clear))
    app.add_handler(MessageHandler(filters.TEXT & ~filters.COMMAND, handle_message))
    app.run_polling()

if __name__ == "__main__":
    main()
